import { useEffect, useRef, useState } from "react";
import { DensityProvider } from "@rain-next/foundations";
import { Grid, Inline, Stack } from "@rain-next/layouts";
import { Button, Checkbox, CheckableTagGroup, IconButton, RadioGroup, Empty, EmptyActions, EmptyDescription, EmptyTitle, Pagination, Select, Table, TableBody, TableCaption, TableCell, TableHead, TableHeader, TableRow, TableScrollArea } from "@rain-next/ui";
import { RefreshIcon } from "@rain-next/ui/icons/refresh";
import { simulateTask } from "../bm-b/state";
import { reportData, summarize } from "./state";
import { TrendChart } from "./trend-chart";
import "./page.css";

function Statistic({ label, value, unit, detail }: { label: string; value: string; unit: string; detail: string }) {
  return <div className="bm-d-statistic" data-rain-id="local-statistic" data-slot="statistic"><dt data-slot="label">{label}</dt><dd><span className="rain-type-data" data-slot="value">{value}</span><span data-slot="unit">{unit}</span></dd><p>{detail}</p></div>;
}
const format = (value: number | null) => value === null ? "—" : value.toFixed(1);

export function LearningReportScenario({ loadTask = simulateTask }: { loadTask?: (signal: AbortSignal, fail: boolean) => Promise<void> }) {
  const [group, setGroup] = useState("全部班级"), [period, setPeriod] = useState("all"), [page, setPage] = useState(1);
  const [metric, setMetric] = useState<"average" | "completion">("average"), [selectedWeek, setSelectedWeek] = useState(6);
  const [dark, setDark] = useState(false), [empty, setEmpty] = useState(false), [failLoad, setFailLoad] = useState(false);
  const [loadState, setLoadState] = useState<"ready" | "loading" | "failed">("ready");
  const requestRef = useRef<AbortController | null>(null), statusRef = useRef<HTMLParagraphElement>(null);
  const firstWeek = period === "recent" ? 4 : 1;
  const rows = empty ? [] : reportData.filter(row => (group === "全部班级" || row.group === group) && row.week >= firstWeek);
  const totals = summarize(rows), weeks = [...new Set(rows.map(row => row.week))], groups = [...new Set(rows.map(row => row.group))];
  const week = weeks.includes(selectedWeek) ? selectedWeek : weeks[weeks.length - 1] ?? 0;
  const unit = metric === "average" ? "分" : "%", metricLabel = metric === "average" ? "平均分" : "提交率";
  const pointValue = (label: string, value: number) => summarize(rows.filter(row => row.group === label && row.week === value))[metric];
  const first = summarize(rows.filter(row => row.week === firstWeek)), last = summarize(rows.filter(row => row.week === 6));
  useEffect(() => () => requestRef.current?.abort(), []);
  useEffect(() => { if (loadState === "failed") statusRef.current?.focus(); }, [loadState]);
  async function reload() {
    if (requestRef.current) return;
    const request = new AbortController(); requestRef.current = request; setLoadState("loading"); setFailLoad(false);
    try {
      await loadTask(request.signal, failLoad);
      if (!request.signal.aborted && requestRef.current === request) setLoadState("ready");
    } catch {
      if (!request.signal.aborted && requestRef.current === request) setLoadState("failed");
    } finally { if (requestRef.current === request && !request.signal.aborted) requestRef.current = null; }
  }
  function changeGroup(value: string) { if (!requestRef.current) { setGroup(value); setPage(1); } }

  return <main className="bm-d" data-rain-id="scenario-bm-d" data-asset-identity="local-experiment" data-rain-theme={dark ? "dark" : "light"}>
    <div className="bm-d-content"><Stack gap={24}>
      <header className="bm-d-header">
        <div><p className="bm-d-eyebrow">教学分析 / 大学物理</p><h1>大学物理 · 学习报告</h1><p>从作业参与和学习表现，了解课堂的变化。</p></div>
        <a href="#bm-c" data-focus="item">进入作业批改</a>
      </header>
      <section className="bm-d-filter" aria-label="统计范围">
        <CheckableTagGroup legend="报告班级" legendHidden size={28} value={group} onValueChange={value => value && changeGroup(value)} disabled={loadState === "loading"}
          options={["全部班级", "一班", "二班"].map(value => ({ value, label: value }))} />
        <Inline gap={12} align="end"><div className="bm-d-period"><Select clearable={false} aria-label="报告周期" disabled={loadState === "loading"} value={period} onValueChange={value => { if (value === "all" || value === "recent") { setPeriod(value); setPage(1); } }} options={[{ value: "all", label: "第 1–6 周" }, { value: "recent", label: "第 4–6 周" }]} /></div>
          <IconButton aria-label="刷新报告" title="刷新报告" icon={<RefreshIcon />} loading={loadState === "loading"} onClick={() => void reload()} />
        </Inline>
      </section>
      <p className="bm-d-scope">统计范围：{group} · 第 {firstWeek}–6 周 · {rows.length} 条周度汇总；指标覆盖全部筛选数据。</p>
      {loadState !== "ready" ? <section className="bm-d-message"><p ref={statusRef} tabIndex={-1} role={loadState === "failed" ? "alert" : "status"}>{loadState === "failed" ? "报告加载失败，当前范围的数据尚未更新。" : "正在更新报告…"}</p>{loadState === "failed" && <Button onClick={() => void reload()}>重试加载</Button>}</section>
        : !rows.length ? <Empty><EmptyTitle>暂无报告数据</EmptyTitle><EmptyDescription>当前范围还没有可用的作业汇总。</EmptyDescription><EmptyActions><Button onClick={() => { setEmpty(false); setPage(1); }}>恢复示例数据</Button></EmptyActions></Empty>
        : <>
          <section aria-label="关键指标"><dl className="bm-d-metrics"><Grid columns={4} minColumnWidth={140} gap={0}>
            <Statistic label="应交作业" value={String(totals.expected)} unit="份" detail="每周应交份数之和" />
            <Statistic label="实际提交" value={String(totals.submissions)} unit="份" detail="每次提交按一份计" />
            <Statistic label="平均分" value={format(totals.average)} unit="分" detail="按提交份数加权 · 满分 100" />
            <Statistic label="提交率" value={`${format(totals.completion)}%`} unit="" detail="实际提交 ÷ 应交作业" />
          </Grid></dl></section>
          <div className="bm-d-report-layout">
          <section className="bm-d-trend" aria-labelledby="bm-d-trend-title"><Stack gap={20}>
            <div className="bm-d-trend-heading"><div><h2 id="bm-d-trend-title">每周学习趋势</h2><p>比较两个班的变化，查看具体周次。</p></div>
              <RadioGroup legend="趋势指标" legendHidden orientation="horizontal" value={metric} onValueChange={value => { if (value === "average" || value === "completion") setMetric(value); }} options={[{ value: "average", label: "平均分" }, { value: "completion", label: "提交率" }]} />
            </div>
            <TrendChart rows={rows} metric={metric} firstWeek={firstWeek} selectedWeek={week} />
            <div className="bm-d-week"><span>按周读数</span><div><Select clearable={false} aria-label="选择趋势周" value={String(week)} onValueChange={value => value && setSelectedWeek(Number(value))} options={weeks.map(value => ({ value: String(value), label: `第 ${value} 周` }))} /></div></div>
            <p className="bm-d-readout" role="status" aria-label="趋势读数">第 {week} 周{metricLabel}：{groups.map(label => `${label} ${format(pointValue(label, week))}${unit}`).join("；")}。</p>
          </Stack></section>
          <div className="bm-d-observations">
            <section className="bm-d-summary" aria-labelledby="bm-d-summary-title"><p className="bm-d-eyebrow">本期观察</p><h2 id="bm-d-summary-title">把变化放回教学现场</h2>
              <p className="rain-type-reading">第 {firstWeek} 周至第 6 周，平均分从 <strong>{format(first.average)} 分</strong>变为 <strong>{format(last.average)} 分</strong>。</p>
              <p className="rain-type-reading">本期共提交 {totals.submissions} 份作业，提交率 {format(totals.completion)}%。可结合每周明细，回看尚未完成的作业。</p>
              <p className="bm-d-scope">周度汇总不含学生身份，无法计算去重人数；趋势不代表教学因果。</p>
            </section>
            <section className="bm-d-insight" aria-labelledby="bm-d-insight-title"><p className="bm-d-ai-label">AI 洞察 · 示例</p><h2 id="bm-d-insight-title">下一次课堂，关注什么？</h2><p className="rain-type-reading">先了解未提交的原因，再从典型作答中选择需要共同讨论的概念。</p><a href="#bm-c" data-focus="item">查看作答与评分</a><p className="bm-d-scope">固定示例建议，未调用 AI 服务。</p></section>
          </div>
          </div>
          <section className="bm-d-details" aria-labelledby="bm-d-details-title"><Inline justify="between"><h2 id="bm-d-details-title">数据明细</h2><span className="bm-d-scope">每页 4 条 · 分页只改变明细</span></Inline>
            <DensityProvider density="compact"><TableScrollArea aria-label="报告明细滚动区"><Table><TableCaption>当前筛选范围的周度汇总，分页不改变关键指标。</TableCaption><TableHeader><TableRow><TableHead>周次</TableHead><TableHead>班级</TableHead><TableHead>应交</TableHead><TableHead>已交</TableHead><TableHead>平均分</TableHead><TableHead>提交率</TableHead></TableRow></TableHeader><TableBody>{rows.slice((page - 1) * 4, page * 4).map(row => <TableRow key={`${row.group}-${row.week}`}><TableCell>第 {row.week} 周</TableCell><TableCell>{row.group}</TableCell><TableCell>{row.expected}</TableCell><TableCell>{row.submitted}</TableCell><TableCell>{format(summarize([row]).average)}</TableCell><TableCell>{format(summarize([row]).completion)}%</TableCell></TableRow>)}</TableBody></Table></TableScrollArea>
            <Pagination total={rows.length} page={page} pageSize={4} showJumper={false} onChange={change => setPage(change.page)} aria-label="报告明细分页" /></DensityProvider>
          </section>
        </>}
      <footer className="bm-d-footer"><p>虚构教学数据 · 仅供报告布局与交互体验 · 字体使用本机已有字体</p><details><summary data-focus="item">演示设置</summary><Inline gap={16}><Button variant="tertiary" onClick={() => setDark(!dark)}>{dark ? "切换浅色" : "切换深色"}</Button><Checkbox label="下次加载失败" disabled={loadState === "loading"} checked={failLoad} onCheckedChange={setFailLoad} /><Button variant="tertiary" disabled={loadState === "loading"} onClick={() => { setEmpty(true); setPage(1); setLoadState("ready"); }}>预览空数据</Button></Inline></details></footer>
    </Stack></div>
  </main>;
}
