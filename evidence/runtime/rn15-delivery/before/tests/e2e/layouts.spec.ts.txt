import { expect, test, type Page } from "@playwright/test";
import { writeFileSync } from "node:fs";

async function check(page: Page, name: string) {
  await page.getByRole("checkbox", { name }).focus(); await page.keyboard.press("Space");
  await expect(page.getByRole("checkbox", { name })).toBeChecked();
}
const split = (page: Page) => page.locator('[data-rain-id="layout-split"]');
const main = (page: Page) => page.getByRole("region", { name: "资料列表", exact: true });
const sidebar = (page: Page) => page.getByRole("region", { name: "资料详情", exact: true });

async function panelGeometry(page: Page) {
  return main(page).evaluate(el => {
    const groups = el.querySelector('[data-example="main-groups"]')!, list = el.querySelector('[data-example="material-list"]')!;
    const information = list.querySelector('[data-example="material-info"]')!;
    const between = (a: Element, b: Element) => b.getBoundingClientRect().top - a.getBoundingClientRect().bottom;
    return { inset: groups.getBoundingClientRect().left - el.getBoundingClientRect().left,
      topInset: groups.getBoundingClientRect().top - el.getBoundingClientRect().top,
      section: between(groups.children[0]!, groups.children[1]!),
      item: between(information.children[0]!, information.children[1]!),
      articles: between(list.children[0]!, list.children[1]!),
      sidebarInset: parseFloat(getComputedStyle(el.nextElementSibling!).paddingTop),
      sidebarGap: getComputedStyle(el.nextElementSibling!.firstElementChild!).gap };
  });
}

async function chooseSpacing(page: Page, name: string, value: number) {
  await page.getByRole("combobox", { name, exact: true }).click();
  await page.getByRole("option", { name: `${value}px`, exact: true }).click();
}

for (const theme of ["light", "dark"] as const) for (const compact of [false, true]) {
  test(`RN-06 ${theme} ${compact ? "compact" : "default"}: actual container geometry, density and independent scroll`, async ({ page }, info) => {
    await page.setViewportSize({ width: 1280, height: 900 }); await page.goto("/#layouts");
    if (theme === "dark") await check(page, "深色主题");
    if (compact) await check(page, "紧凑控件");
    expect(await page.getByRole("heading", { name: "布局与组合", exact: true }).evaluate(el => getComputedStyle(el).color === getComputedStyle(el.parentElement!).color)).toBe(true);
    const grid = page.locator('[data-example="settings-grid"]'), root = split(page);
    await expect(root).toHaveAttribute("data-stacked", "false");
    await expect(grid).toHaveCSS("gap", "12px");
    expect((await grid.evaluate(el => getComputedStyle(el).gridTemplateColumns)).split(" ")).toHaveLength(2);
    const fields = await grid.locator(':scope > *').evaluateAll(nodes => nodes.map(el => { const r = el.getBoundingClientRect(); return { x: r.x, y: r.y, right: r.right }; }));
    expect(fields[1]!.x - fields[0]!.right).toBe(12); expect(fields[1]!.y).toBe(fields[0]!.y);
    const labels = await grid.locator('[data-slot="label"]').evaluateAll(nodes => nodes.map(el => el.getBoundingClientRect().y));
    expect(labels[1]).toBe(labels[0]);
    const courseControl = await grid.locator('[data-rain-id="ui-input"]').boundingBox();
    const modeControl = await grid.locator('[data-rain-id="ui-select"]').boundingBox();
    expect(modeControl!.y).toBe(courseControl!.y);
    const spacing = await panelGeometry(page);
    expect(spacing).toEqual({ inset: 16, topInset: 16, section: 12, item: 8, articles: 12, sidebarInset: 16, sidebarGap: "12px" });
    await expect(page.locator('[data-example="layout-settings"]')).toHaveCSS("padding-top", "20px");
    await expect(page.locator('[data-example="settings-sections"]')).toHaveCSS("gap", "16px");
    await expect(page.locator('[data-example="material-sections"]')).toHaveCSS("gap", "16px");
    await expect(page.locator('.layouts-review')).toHaveCSS("padding-top", "32px");
    await expect(page.locator('.layouts-review-content')).toHaveCSS("gap", "24px");
    await expect(sidebar(page)).toHaveCSS("width", "320px");
    await expect(page.getByRole("textbox", { name: "课程名称", exact: true }).locator('xpath=ancestor::*[@data-rain-id="ui-input"]')).toHaveAttribute("data-size", compact ? "24" : "32");
    await expect(page.getByRole("button", { name: "保存设置", exact: true })).toHaveCSS("height", "32px");
    // Constrain a real consumer to exercise both scroll owners even when short content fits.
    await root.evaluate(el => { el.style.height = "360px"; });
    for (const pane of [main(page), sidebar(page)]) {
      await expect(pane).toHaveCSS("overflow-y", "auto");
      expect(await pane.evaluate(el => el.scrollHeight > el.clientHeight)).toBe(true);
    }
    await main(page).evaluate(el => { el.scrollTop = 180; });
    expect(await sidebar(page).evaluate(el => el.scrollTop)).toBe(0);
    await sidebar(page).evaluate(el => { el.scrollTop = 100; });
    expect(await main(page).evaluate(el => el.scrollTop)).toBe(180);
    await main(page).evaluate(el => { el.scrollTop = 0; }); await sidebar(page).evaluate(el => { el.scrollTop = 0; });
    await root.evaluate(el => { el.style.removeProperty("height"); });
    const facts = await root.evaluate(el => ({ width: el.getBoundingClientRect().width, gap: getComputedStyle(el).gap, stacked: el.getAttribute("data-stacked"), parts: [...el.children].map(node => ({ slot: node.getAttribute("data-slot"), width: node.getBoundingClientRect().width, height: node.getBoundingClientRect().height })) }));
    writeFileSync(info.outputPath(`wide-${theme}-${compact}.json`), JSON.stringify({ facts, spacing }, null, 2));
    await page.screenshot({ path: info.outputPath(`wide-${theme}-${compact}.png`), fullPage: true });
    await check(page, "窄容器");
    await expect(root).toHaveAttribute("data-stacked", "true");
    expect((await grid.evaluate(el => getComputedStyle(el).gridTemplateColumns)).split(" ")).toHaveLength(1);
    const before = await main(page).boundingBox(), after = await sidebar(page).boundingBox();
    expect(after!.y - (before!.y + before!.height)).toBe(12);
    expect(after!.width).toBe(before!.width);
    expect(await root.evaluate(el => el.scrollWidth <= el.clientWidth)).toBe(true);
    const stackedSpacing = await panelGeometry(page);
    expect(stackedSpacing.section).toBe(12); expect(stackedSpacing.item).toBe(8); expect(stackedSpacing.articles).toBe(12);
    await page.screenshot({ path: info.outputPath(`narrow-container-${theme}-${compact}.png`), fullPage: true });
  });
}

test("spacing hierarchy: independent choices, equality and zero preserve live drafts and actual panel geometry", async ({ page }, info) => {
  await page.setViewportSize({ width: 1280, height: 900 }); await page.goto("/#layouts");
  const title = page.getByRole("textbox", { name: "资料标题", exact: true }), original = await title.elementHandle();
  await title.fill("调整间距时保留的未保存草稿");
  await page.getByText("调整资料面板间距", { exact: true }).click();
  const results = [];
  for (const [inset, section, item] of [[24, 20, 12], [20, 20, 12], [16, 16, 16], [40, 36, 24], [0, 0, 0]]) {
    await chooseSpacing(page, "边缘留白", inset!); await chooseSpacing(page, "分区间距", section!); await chooseSpacing(page, "信息间距", item!);
    await page.getByRole("button", { name: "应用面板间距" }).click();
    await expect(page.getByRole("alert")).toHaveCount(0);
    const geometry = await panelGeometry(page); results.push({ chosen: { inset, section, item }, geometry });
    expect(geometry).toEqual({ inset, topInset: inset, section, item, articles: section, sidebarInset: inset, sidebarGap: `${section}px` });
    await expect(title).toHaveValue("调整间距时保留的未保存草稿");
    expect(await title.evaluate((el, prior) => el === prior, original)).toBe(true);
  }
  await chooseSpacing(page, "边缘留白", 16); await chooseSpacing(page, "分区间距", 16); await chooseSpacing(page, "信息间距", 16);
  await page.getByRole("button", { name: "应用面板间距" }).click();
  await page.setViewportSize({ width: 375, height: 800 }); await check(page, "深色主题"); await check(page, "紧凑控件");
  await expect(split(page)).toHaveAttribute("data-stacked", "true");
  expect((await panelGeometry(page)).section).toBe(16);
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)).toBe(true);
  await page.screenshot({ path: info.outputPath("equal-375-dark-compact.png"), fullPage: true });
  writeFileSync(info.outputPath("spacing-choices.json"), JSON.stringify(results, null, 2));
});

test("outer cards: apply all three spacings independently from panels without losing drafts", async ({ page }, info) => {
  await page.setViewportSize({ width: 1280, height: 900 }); await page.goto("/#layouts");
  const course = page.getByRole("textbox", { name: "课程名称", exact: true }), title = page.getByRole("textbox", { name: "资料标题", exact: true });
  const originalCourse = await course.elementHandle(), originalTitle = await title.elementHandle();
  await course.fill("外层卡片调整保留课程草稿"); await title.fill("外层卡片调整保留资料草稿");
  await page.getByText("调整外层卡片间距", { exact: true }).click();
  const form = page.getByRole("form", { name: "外层卡片间距", exact: true });
  async function setCard(inset: number, section: number, item: number) {
    for (const [name, value] of [["边缘留白", inset], ["分区间距", section], ["信息间距", item]] as const) {
      await form.getByRole("combobox", { name, exact: true }).click();
      await page.getByRole("option", { name: `${value}px`, exact: true }).click();
    }
    await form.getByRole("button", { name: "应用卡片间距" }).click();
  }
  const cards = page.locator('[data-example="layout-settings"], [data-example="layout-materials"]');
  const results = [];
  for (const [inset, section, item] of [[32, 24, 16], [16, 16, 16], [0, 0, 0], [40, 40, 40]] as const) {
    await setCard(inset, section, item); await expect(form.getByRole("alert")).toHaveCount(0);
    const geometry = await cards.evaluateAll(nodes => nodes.map(el => {
      const header = el.querySelector(':scope > [data-slot="header"]')!, content = el.querySelector(':scope > [data-slot="content"]')!;
      const edge = el.getBoundingClientRect(), h = header.getBoundingClientRect(), c = content.getBoundingClientRect(), border = parseFloat(getComputedStyle(el).borderWidth);
      const groups = content.querySelector('[data-example="settings-sections"], [data-example="material-sections"]')!;
      const info = groups.querySelector('[data-example="settings-grid"], [data-example="query-inline"]')!;
      return { top: h.top - edge.top - border, bottom: edge.bottom - c.bottom - border,
        headerLeft: header.firstElementChild!.getBoundingClientRect().left - edge.left - border,
        contentLeft: groups.getBoundingClientRect().left - edge.left - border,
        contentRight: edge.right - groups.getBoundingClientRect().right - border,
        headerSection: c.top - h.bottom, bodySection: groups.children[1]!.getBoundingClientRect().top - groups.children[0]!.getBoundingClientRect().bottom,
        item: getComputedStyle(info).gap };
    }));
    expect(geometry).toEqual([0, 1].map(() => ({ top: inset, bottom: inset, headerLeft: inset, contentLeft: inset, contentRight: inset, headerSection: section, bodySection: section, item: `${item}px` })));
    const smallGaps = await cards.evaluateAll(nodes => nodes.flatMap(el => [
      parseFloat(getComputedStyle(el.querySelector(':scope > [data-slot="header"]')!).rowGap),
      parseFloat(getComputedStyle(el.querySelector('.layout-example-actions')!).gap),
    ]));
    for (const gap of smallGaps) expect(gap).toBeLessThanOrEqual(item);
    expect((await panelGeometry(page)).inset).toBe(16);
    expect((await panelGeometry(page)).item).toBe(8);
    expect(await course.evaluate((el, old) => el === old, originalCourse)).toBe(true);
    expect(await title.evaluate((el, old) => el === old, originalTitle)).toBe(true);
    await expect(course).toHaveValue("外层卡片调整保留课程草稿"); await expect(title).toHaveValue("外层卡片调整保留资料草稿");
    results.push({ chosen: { inset, section, item }, geometry });
  }
  await setCard(8, 16, 8); await expect(form.getByRole("alert")).toBeVisible();
  await expect(cards.first()).toHaveCSS("padding-top", "40px");
  await setCard(16, 8, 12); await expect(form.getByRole("alert")).toBeVisible();
  await expect(cards.first()).toHaveCSS("gap", "40px");
  await setCard(40, 40, 40); await expect(form.getByRole("alert")).toHaveCount(0);
  await page.getByText("调整外层卡片间距", { exact: true }).click();
  await page.getByText("调整资料面板间距", { exact: true }).click();
  await chooseSpacing(page, "边缘留白", 24); await page.getByRole("button", { name: "应用面板间距" }).click();
  await expect(cards.first()).toHaveCSS("padding-top", "40px"); expect((await panelGeometry(page)).inset).toBe(24);
  await page.setViewportSize({ width: 320, height: 800 }); await check(page, "深色主题"); await check(page, "紧凑控件");
  await expect(split(page)).toHaveAttribute("data-stacked", "true");
  expect(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth)).toBe(true);
  const buttonsFit = await cards.evaluateAll(nodes => nodes.every(el => {
    const content = el.querySelector(':scope > [data-slot="content"]')!, bounds = content.getBoundingClientRect(), css = getComputedStyle(content);
    return [...el.querySelectorAll('.layout-example-actions button')].every(button => {
      const r = button.getBoundingClientRect();
      return r.left >= bounds.left + parseFloat(css.paddingLeft) && r.right <= bounds.right - parseFloat(css.paddingRight);
    });
  }));
  expect(buttonsFit).toBe(true);
  await page.screenshot({ path: info.outputPath("outer-40-panel-24-narrow.png"), fullPage: true });
  writeFileSync(info.outputPath("outer-card-spacing.json"), JSON.stringify(results, null, 2));
});
