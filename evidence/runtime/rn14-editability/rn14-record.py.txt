from pathlib import Path
import datetime, difflib, hashlib, json, shutil, subprocess, sys

source=Path('/tmp/rain-next-rn14-20260915-1817')
evidence=Path('/Users/wenchengkai/CodexFile/雨课堂-shadcn组件库架构/.worktrees/card-v2/evidence/runtime/rn14-editability')
def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(path, value): path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
def inventory():
 result={}
 for folder in ('apps','packages'):
  for path in sorted((source/folder).rglob('*')):
   if path.is_file() and not any(x in path.parts for x in ('node_modules','dist')):
    result[str(path.relative_to(source))]=hashlib.sha256(path.read_bytes()).hexdigest()
 return result
def copyapp(folder):
 for file in inventory():
  if file.startswith('apps/'):
   target=folder/(file+'.txt');target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes((source/file).read_bytes())
action, name = sys.argv[1:3]; directory=evidence/name
if action=='begin':
 directory.mkdir(); files=inventory(); copyapp(directory/'before')
 save(directory/'input.json',{'startedAt':now(),'inputCommit':json.loads((source/'source-lock.json').read_text())['inputCommit'],'files':files,'inputSnapshotSha256':hashlib.sha256(json.dumps(files,sort_keys=True).encode()).hexdigest()})
elif action=='run':
 label=sys.argv[3]; command=sys.argv[4:]; started=now()
 if label=='browser': shutil.copy(source/'rn14-browser.mjs',directory/'browser-check.mjs.txt')
 with (directory/(label+'.log')).open('w') as log: result=subprocess.run(command,cwd=source,stdout=log,stderr=subprocess.STDOUT)
 save(directory/(label+'-command.json'),{'command':command,'cwd':str(source),'startedAt':started,'finishedAt':now(),'exitCode':result.returncode})
 print(label, result.returncode);sys.exit(result.returncode)
elif action=='end':
 before=json.loads((directory/'input.json').read_text());after=inventory(); changed=[f for f in sorted(set(before['files'])|set(after)) if before['files'].get(f)!=after.get(f)]
 allowed=sys.argv[3:]; assert changed and set(changed)<=set(allowed),(changed,allowed)
 assert not any(f.startswith('packages/') for f in changed)
 copyapp(directory/'after'); diff=[]

  diff+=list(difflib.unified_diff(old.read_text().splitlines(True) if old.exists() else [],new.read_text().splitlines(True),fromfile='a/'+file,tofile='b/'+file))
 (directory/'changes.diff').write_text(''.join(diff))
 ended=now();elapsed=(datetime.datetime.fromisoformat(ended)-datetime.datetime.fromisoformat(before['startedAt'])).total_seconds()
 save(directory/'result.json',{'startedAt':before['startedAt'],'endedAt':ended,'wallClockSecondsIncludingControllerValidation':elapsed,'inputCommit':before['inputCommit'],'inputSnapshotSha256':before.get('inputSnapshotSha256',hashlib.sha256(json.dumps(before['files'],sort_keys=True).encode()).hexdigest()),'outputSnapshotSha256':hashlib.sha256(json.dumps(after,sort_keys=True).encode()).hexdigest(),'changedFiles':changed,'linesAdded':sum(x.startswith('+') and not x.startswith('+++') for x in diff),'linesRemoved':sum(x.startswith('-') and not x.startswith('---') for x in diff),'sharedFilesUnchanged':True,'nonTargetFilesUnchanged':True,'modelExactId':'not-measured','tokenUsage':'not-measured','humanCorrectionPrompts':'not-measured','efficiencyImprovement':'not-measured','unmeasuredReason':'No provider usage telemetry or real-user correction/comparison arm was measured.','files':after})
 agent=Path('/tmp/rn14-'+name+'-agent.json')
 if agent.exists():shutil.copy(agent,directory/'agent-report.json')
 print(name,changed)
